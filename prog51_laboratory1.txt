#include <iostream>

using namespace std;

int main() {
   
    double scores[5];
    
    cout << "Enter 5 scores:\n";
    
    
    for (int i = 0; i < 5; i++) {
        cout << "Enter score " << (i + 1) << ": ";
        cin >> scores[i];
    }
    
  
    double highest = scores[0];
    
   
    for (int i = 1; i < 5; i++) {
        if (scores[i] > highest) {
            highest = scores[i];
        }
    }
    
   
    cout << "\nThe highest score is: " << highest << endl;
    
    return 0;
}
